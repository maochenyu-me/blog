# 外链中转安全页（SafeGo）实施计划

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 在 Nuxt 4 博客上实现外链中转安全页：站外链接统一先进入 `/go` 提示页，读者可「直接跳转」或「回到上页」（回到原页面并恢复滚动位置），跳转时长、生效页面、生效外链与全部文案都由 `app/data/safego.ts` 配置。

**Architecture:** 一个挂在 `document` **捕获阶段**的客户端插件接管站外链接点击（NuxtLink 对站外链接渲染的是裸 `<a target="_blank">`，其 `onClick` 首句即 `return`，不存在 JS 导航，所以 `preventDefault()` 就够用且无竞态）。所有判定逻辑抽到 `shared/utils/safego.ts` 的纯函数层，零第三方依赖，可直接被 `node --test` 导入。`/go` 用专用极简布局渲染；跳出用 `location.replace()`，让历史栈里不残留中转页，从而在外部站点按返回键直达文章原位置。

**Tech Stack:** Nuxt 4.5.2 / Vue 3 / Nuxt Content 3 / 原生 CSS + postcss-nesting / @vueuse/nuxt / `node:test`（Node 24 类型擦除）

**设计依据:** `docs/superpowers/specs/2026-11-02-safego-design.md`（已评审通过）

```ts
export type SafeGoLinkMode = 'all' | 'listed'

/** 可疑目标的特征类型，键名对应 app/data/safego.ts 的 text.risks */
export type SafeGoRiskKind = 'http' | 'ip' | 'punycode' | 'long' | 'port'

export interface SafeGoText {
	/** 中转页浏览器标题 */
	title: string
	/** 主提示语，`{site}` 会被替换成站点名 */
	leave: string
	/** 倒计时提示语，`{seconds}` 会被替换成剩余秒数 */
	countdown: string
	/** countdown 为负数时的提示语 */
	manual: string
	/** 主按钮文案 */
	jump: string
	/** 次按钮文案 */
	back: string
	/** 参数缺失/非法时的标题 */
	invalid: string
	/** 参数缺失/非法时的说明 */
	invalidDesc: string
	/** 失败态的按钮文案 */
	home: string
	risks: Record<SafeGoRiskKind, string>
}

export interface SafeGoConfig {
	/** 总开关：false 时插件不注册任何监听，全站外链恢复直连 */
	enable: boolean
	route: {
		/** 中转页路由，需与 app/pages 下的文件名一致 */
		path: string
		/** 目标地址的查询参数名 */
		param: string
		/** 是否用 base64url 编码目标地址 */
		base64: boolean
	}
	pages: {
		/** 生效路径，glob 风格；`/**` 表示全站 */
		include: string[]
		/** 排除路径，glob 风格 */
		exclude: string[]
	}
	links: {
		/** all = 除白名单外全部中转；listed = 只中转 domains 里列出的域名 */
		mode: SafeGoLinkMode
		/** 域名名单，按主机后缀匹配 */
		domains: string[]
		/** 带这些属性的链接一律不中转 */
		ignoreAttrs: string[]
	}
	behavior: {
		/** 自动跳转倒计时秒数；负数表示不自动跳转 */
		countdown: number
		/** 悬停或键盘聚焦卡片时暂停倒计时 */
		pauseOnInteraction: boolean
		/** 切换浏览器标签页时暂停倒计时 */
		pauseWhenHidden: boolean
		/** Ctrl/Cmd/Shift + 左键、鼠标中键是否也走中转页 */
		interceptModifiedClick: boolean
	}
	text: SafeGoText
}

/** 中转页对目标地址的解析结果 */
export type SafeGoTarget =
	| { ok: true, href: string, host: string, risks: SafeGoRiskKind[] }
	| { ok: false, reason: 'missing' | 'malformed' | 'protocol' }
```

```ts
/**
 * 外链中转安全页配置
 *
 * 改动本文件即改变全站外链行为，无需触碰任何组件代码。
 * 中转页本身的路由在 app/pages/go.vue，视觉在 app/layouts/safego.vue。
 */
export const safeGoConfig: SafeGoConfig = {
	/** 总开关：false 时插件不注册任何监听，全站外链恢复直连 */
	enable: true,

	/** 中转页路由与传参 */
	route: {
		/** 中转页路由，需与 app/pages 下的文件名一致 */
		path: '/go',
		/** 目标地址的查询参数名，如 /go?u=https%3A%2F%2Fexample.com */
		param: 'u',
		/**
		 * 是否用 base64url 编码目标地址。
		 * true 时地址栏不暴露原始 URL，但读者也无法在地址栏确认目标；
		 * 注意这**不是**安全措施，只是观感选择。
		 */
		base64: false,
	},

	/** 哪些页面生效：全站生效，只排除下面这些 */
	pages: {
		/** 生效路径，glob 风格；`/**` 表示全站 */
		include: ['/**'],
		/**
		 * 排除路径，glob 风格。中转页自身必须排除，否则会死循环。
		 * 需要豁免其他页面时在这里追加，例如：
		 *   '/link'   友链页（若不想让访客跳转友链时多一步）
		 *   '/about'  关于页

	links: {

		ignoreAttrs: ['download', 'data-safego-ignore'],
	},

	/** 跳转行为 */
	behavior: {
		/** 自动跳转倒计时秒数；**负数表示不自动跳转**，只保留手动按钮 */
		countdown: 5,
		/** 鼠标悬停或键盘聚焦到卡片上时暂停倒计时，避免读者没看完就被跳走 */
		pauseOnInteraction: true,
		/** 切换浏览器标签页时暂停倒计时 */
		pauseWhenHidden: true,
		/** Ctrl/Cmd/Shift + 左键、鼠标中键是否也走中转页（在新标签页打开中转页） */
		interceptModifiedClick: true,
	},

	/** 文案，全部可覆写；`{site}` 与 `{seconds}` 为占位符 */
	text: {
		/** 中转页浏览器标题 */
		title: '链接安全提示',
		/** 主提示语，`{site}` 会被替换成站点名 */
		leave: '你正在离开「{site}」',
		/** 倒计时提示语，`{seconds}` 会被替换成剩余秒数 */
		countdown: '{seconds} 秒后自动跳转，请自行确认链接安全性',

		manual: '请确认链接安全后手动跳转',
		/** 主按钮文案 */
		jump: '直接跳转',
		/** 次按钮文案 */
		back: '回到上页',
		/** 参数缺失/非法时的标题 */
		invalid: '这个中转链接无效',
		/** 参数缺失/非法时的说明 */
		invalidDesc: '目标地址缺失，或使用了不被允许的协议。为安全起见已阻止跳转。',
		/** 失败态的按钮文案 */
		home: '返回首页',
		/** 风险特征提示文案，键名对应 app/types/safego.ts 的 SafeGoRiskKind */
```

**Step 1: 先写失败测试**

创建 `scripts/safego/safego.test.ts`：

```ts
/**
 * 外链中转安全页纯函数回归用例
 *
 * 放在 scripts/ 下是为了被仓库既有的 `pnpm test`
 * （`node --test "scripts/**\/*.test.ts"`）直接收进来，不必改 package.json。
 * 本文件只依赖 node:* 与仓库内的 .ts 模块。
 */
import assert from 'node:assert/strict'
import { describe, it } from 'node:test'
import {
	buildSafeGoUrl,
	detectRisks,
	isPageInScope,
	matchDomainHost,
	matchPathPattern,
	resolveInterceptedHref,
	resolveTarget,
} from '../../shared/utils/safego.ts'

const includeAll = { include: ['/**'], exclude: ['/go', '/previews/**'] }
const whitelistAll = { mode: 'all' as const, domains: ['518339.xyz', 'github.com'], ignoreAttrs: [] }
const whitelistListed = { mode: 'listed' as const, domains: ['github.com'], ignoreAttrs: [] }
const BASE = 'https://blog.kemeow.top/2026/example'

describe('matchPathPattern', () => {
	it('`/**` 命中任意路径', () => {
		assert.equal(matchPathPattern('/**', '/'), true)
		assert.equal(matchPathPattern('/**', '/posts/a'), true)
	})

	it('精确路径忽略尾部斜杠', () => {
		assert.equal(matchPathPattern('/go', '/go'), true)
		assert.equal(matchPathPattern('/go', '/go/'), true)
		assert.equal(matchPathPattern('/go', '/go2'), false)
	})

	it('结尾的 `/**` 允许整段后缀缺失', () => {
		assert.equal(matchPathPattern('/previews/**', '/previews/a/b'), true)
		assert.equal(matchPathPattern('/previews/**', '/previews'), true)
		assert.equal(matchPathPattern('/previews/**', '/other'), false)
	})

	it('`*` 不跨路径段', () => {
		assert.equal(matchPathPattern('/post/*', '/post/a'), true)
		assert.equal(matchPathPattern('/post/*', '/post/a/b'), false)
	})
})

describe('isPageInScope', () => {
	it('include 命中且未被 exclude 命中时为真', () => {
		assert.equal(isPageInScope('/2026/example', includeAll), true)
	})

	it('exclude 优先于 include', () => {
		assert.equal(isPageInScope('/go', includeAll), false)
		assert.equal(isPageInScope('/previews/draft', includeAll), false)
	})

	it('include 未命中时为假', () => {
		assert.equal(isPageInScope('/2026/example', { include: ['/posts/**'], exclude: [] }), false)
	})
})

describe('matchDomainHost', () => {
	it('精确命中', () => {
		assert.equal(matchDomainHost('github.com', ['github.com']), true)
	})

	it('子域命中', () => {
		assert.equal(matchDomainHost('gist.github.com', ['github.com']), true)
		assert.equal(matchDomainHost('blog.518339.xyz', ['518339.xyz']), true)
	})

	it('伪造后缀不命中', () => {
		assert.equal(matchDomainHost('evilgithub.com', ['github.com']), false)
		assert.equal(matchDomainHost('github.com.evil.net', ['github.com']), false)
	})

	it('大小写与尾点不敏感', () => {
		assert.equal(matchDomainHost('GITHUB.COM', ['github.com']), true)
		assert.equal(matchDomainHost('github.com.', ['github.com']), true)
	})
})

describe('resolveInterceptedHref', () => {
	it('站外 https 链接被接管并规范化', () => {
		assert.equal(resolveInterceptedHref('https://example.com/a', BASE, whitelistAll), 'https://example.com/a')
	})

	it('协议相对链接按当前协议解析', () => {
		assert.equal(resolveInterceptedHref('//example.com/a', BASE, whitelistAll), 'https://example.com/a')
	})

	it('站内链接放行', () => {
		assert.equal(resolveInterceptedHref('/posts/a', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('https://blog.kemeow.top/x', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('#hash', BASE, whitelistAll), null)
	})

	it('非 http(s) 协议放行', () => {
		assert.equal(resolveInterceptedHref('mailto:a@b.c', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('tel:+8613800000000', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('javascript:alert(1)', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('data:text/html,x', BASE, whitelistAll), null)
	})

	it('mode=all 时 domains 是豁免名单', () => {
		assert.equal(resolveInterceptedHref('https://github.com/x', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('https://gist.github.com/x', BASE, whitelistAll), null)
		assert.equal(resolveInterceptedHref('https://other.com/x', BASE, whitelistAll), 'https://other.com/x')
	})

	it('mode=listed 时 domains 是命中名单', () => {
		assert.equal(resolveInterceptedHref('https://github.com/x', BASE, whitelistListed), 'https://github.com/x')
		assert.equal(resolveInterceptedHref('https://other.com/x', BASE, whitelistListed), null)
	})
})

describe('resolveTarget', () => {
	it('缺参报 missing', () => {
		assert.deepEqual(resolveTarget(undefined, false), { ok: false, reason: 'missing' })
		assert.deepEqual(resolveTarget('', false), { ok: false, reason: 'missing' })
	})

	it('正常 https 目标通过', () => {
		const result = resolveTarget('https://example.com/a?b=1', false)
		assert.equal(result.ok, true)
		assert.equal(result.ok && result.host, 'example.com')
	})

	it('只放行 http 与 https，其余报 protocol', () => {
		assert.deepEqual(resolveTarget('javascript:alert(1)', false), { ok: false, reason: 'protocol' })
		assert.deepEqual(resolveTarget('ftp://example.com/a', false), { ok: false, reason: 'protocol' })
		assert.deepEqual(resolveTarget('data:text/html,x', false), { ok: false, reason: 'protocol' })
	})

	it('无法解析的地址报 malformed', () => {
		assert.deepEqual(resolveTarget('not a url', false), { ok: false, reason: 'malformed' })
		assert.deepEqual(resolveTarget('aHR0cHM6', true), { ok: false, reason: 'malformed' })
	})

	it('不再重复解码：百分号转义按原样保留', () => {
		const result = resolveTarget('https://example.com/a%20b', false)
		assert.equal(result.ok && result.href, 'https://example.com/a%20b')
	})

	it('base64url 往返，含中文', () => {
		const target = 'https://example.com/搜索?q=你好 世界'
		const url = buildSafeGoUrl('/go', 'u', target, true)
		const raw = new URLSearchParams(url.slice(url.indexOf('?'))).get('u')!
		const result = resolveTarget(raw, true)
		assert.equal(result.ok && result.href, new URL(target).href)
		assert.equal(raw.includes('+'), false)
		assert.equal(raw.includes('='), false)
	})
})

describe('detectRisks', () => {
	it('干净的 https 地址没有风险提示', () => {
		assert.deepEqual(detectRisks(new URL('https://example.com/a')), [])
	})

	it('http 明文', () => {
		assert.deepEqual(detectRisks(new URL('http://example.com/a')), ['http'])
	})

	it('IPv4 / IPv6 字面量', () => {
		assert.deepEqual(detectRisks(new URL('http://127.0.0.1/a')), ['http', 'ip'])
		assert.deepEqual(detectRisks(new URL('http://[::1]/a')), ['http', 'ip'])
	})

	it('punycode 域名', () => {
		assert.deepEqual(detectRisks(new URL('https://xn--fiq228c.com/')), ['punycode'])
	})

	it('非标准端口', () => {
		assert.deepEqual(detectRisks(new URL('https://example.com:8443/a')), ['port'])
		assert.deepEqual(detectRisks(new URL('https://example.com:443/a')), [])
	})

	it('超长地址', () => {
		assert.deepEqual(detectRisks(new URL(`https://example.com/${'a'.repeat(200)}`)), ['long'])
	})
})

describe('buildSafeGoUrl', () => {
	it('普通模式把目标地址编码进查询串且能原样取回', () => {
		const url = buildSafeGoUrl('/go', 'u', 'https://example.com/a?b=1&c=2', false)
		assert.equal(url.startsWith('/go?u='), true)
		const raw = new URLSearchParams(url.slice(url.indexOf('?'))).get('u')
		assert.equal(raw, 'https://example.com/a?b=1&c=2')
	})

	it('参数名可配', () => {
		assert.equal(buildSafeGoUrl('/jump', 'target', 'https://a.com/', false).startsWith('/jump?target='), true)
	})
})
```

```ts
/**
 * 外链中转安全页的纯判定层
 *
 * 这里只做与宿主环境无关的计算：不读写 document、不依赖任何第三方包，
 * 因此既可以被浏览器端的拦截器与中转页复用，也能被 node --test 直接导入。
 */
import type { SafeGoConfig, SafeGoRiskKind, SafeGoTarget } from '~/types/safego'

/** 允许跳转的协议白名单；中转页只认这两个，其余一律拒绝 */
const ALLOWED_PROTOCOLS = new Set(['http:', 'https:'])
/** 默认端口；其余端口视为可疑 */
const DEFAULT_PORTS = new Set(['', '80', '443'])
/** 超过这个长度视为可疑地址 */
const SUSPICIOUS_URL_LENGTH = 160
const IPV4_PATTERN = /^\d{1,3}(?:\.\d{1,3}){3}$/

/** 去掉尾部斜杠，让 `/go/` 与 `/go` 等价；空串归一成 `/` */
function normalizePath(path: string) {
	return path.replace(/\/+$/, '') || '/'
}

/** 把 glob 模式编译成正则：`**` 跨路径段，`*` 只在单个路径段内匹配 */
function globToRegExp(pattern: string) {
	let source = ''
	for (let index = 0; index < pattern.length; index++) {
		const char = pattern[index]
		if (char !== '*') {
			source += /[.+^${}()|[\]\\]/.test(char) ? `\\${char}` : char
			continue
		}
		if (pattern[index + 1] === '*') {
			index += 1
			source += '.*'
			continue
		}
		source += '[^/]*'
	}
	// 结尾的 `/**` 允许整段后缀缺失，让 `/previews/**` 同时命中 `/previews`
	if (source.length > 3 && source.endsWith('/.*'))
		source = `${source.slice(0, -3)}(?:/.*)?`
	return new RegExp(`^${source}$`)
}

/** 页面路径是否命中 glob 模式 */
export function matchPathPattern(pattern: string, path: string) {
	return globToRegExp(normalizePath(pattern)).test(normalizePath(path))
}

/** 当前页面是否在中转页的生效范围内 */
export function isPageInScope(path: string, pages: SafeGoConfig['pages']) {
	if (!pages.include.some(pattern => matchPathPattern(pattern, path)))
		return false
	return !pages.exclude.some(pattern => matchPathPattern(pattern, path))
}

/** 主机名后缀匹配：`github.com` 命中 github.com 与 gist.github.com，但不命中 evilgithub.com */
export function matchDomainHost(host: string, domains: string[]) {
	const normalized = host.toLowerCase().replace(/\.$/, '')
	return domains.some((domain) => {
		const target = domain.toLowerCase().replace(/\.$/, '')
		return normalized === target || normalized.endsWith(`.${target}`)
	})
}

/**
 * 判断一个锚点 href 是否应该被中转，命中则返回规范化后的目标地址。
 *
 * 返回 null 表示放行（站内链接、非 http(s) 协议、命中名单规则之外）。
 */
export function resolveInterceptedHref(href: string, base: string, links: SafeGoConfig['links']) {
	let baseUrl: URL
	try {
		baseUrl = new URL(base)
	}
	catch {
		return null
	}
	let url: URL
	try {
		url = new URL(href, baseUrl)
	}
	catch {
		return null
	}
	if (!ALLOWED_PROTOCOLS.has(url.protocol))
		return null
	if (url.origin === baseUrl.origin)
		return null
	const listed = matchDomainHost(url.hostname, links.domains)
	// mode=all 时 domains 是豁免名单；mode=listed 时 domains 是命中名单
	if (links.mode === 'all' ? listed : !listed)
		return null
	return url.href
}

/**
 * 解析中转页参数。
 *
 * 注意：查询参数在 Vue Router 里已经解过一次码，这里**不能再解码**，
 * 否则目标地址里的百分号转义会被二次破坏；只有 base64 模式才需要自己解码。
 */
export function resolveTarget(raw: string | undefined, base64: boolean): SafeGoTarget {
	if (!raw)
		return { ok: false, reason: 'missing' }
	const decoded = base64 ? decodeBase64Url(raw) : raw
	if (decoded === null)
		return { ok: false, reason: 'malformed' }
	let url: URL
	try {
		url = new URL(decoded)
	}
	catch {
		return { ok: false, reason: 'malformed' }
	}
	if (!ALLOWED_PROTOCOLS.has(url.protocol))
		return { ok: false, reason: 'protocol' }
	return { ok: true, href: url.href, host: url.host, risks: detectRisks(url) }
}

/** 目标地址的可疑特征；顺序固定，便于展示与测试 */
export function detectRisks(url: URL): SafeGoRiskKind[] {
	const risks: SafeGoRiskKind[] = []
	if (url.protocol === 'http:')
		risks.push('http')
	const host = url.hostname.toLowerCase()
	if (IPV4_PATTERN.test(host) || (host.startsWith('[') && host.endsWith(']')))
		risks.push('ip')
	if (host.split('.').some(label => label.startsWith('xn--')))
		risks.push('punycode')
	if (url.href.length > SUSPICIOUS_URL_LENGTH)
		risks.push('long')
	if (!DEFAULT_PORTS.has(url.port))
		risks.push('port')
	return risks
}

/** 构造中转页地址 */
export function buildSafeGoUrl(path: string, param: string, target: string, base64: boolean) {
	const value = base64 ? encodeBase64Url(target) : target
	return `${path}?${new URLSearchParams({ [param]: value }).toString()}`
}

function encodeBase64Url(value: string) {
	const bytes = new TextEncoder().encode(value)
	let binary = ''
	for (const byte of bytes)
		binary += String.fromCodePoint(byte)
	return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '')
}

/** base64url 解码，同时容忍标准 base64 与缺失的填充；失败返回 null */
function decodeBase64Url(value: string) {
	const normalized = value.replace(/-/g, '+').replace(/_/g, '/')
	const padded = normalized.padEnd(Math.ceil(normalized.length / 4) * 4, '=')
	try {
		const binary = atob(padded)
		const bytes = Uint8Array.from(binary, char => char.codePointAt(0) ?? 0)
		return new TextDecoder().decode(bytes)
	}
	catch {
		return null
	}
}
```

```ts
import { safeGoConfig } from '~/data/safego'

/**
 * 外链中转：在 document 捕获阶段接管站外链接的点击。
 *
 * 为什么用捕获阶段就够了：`NuxtLink` 在 `isExternal || hasTarget` 时渲染的是裸
 * `<a target="_blank">`，它的 onClick 第一句就是
 * `if (isExternal.value || hasTarget.value) return`（nuxt-link.js:324-325），
 * 站外链接根本不存在 JS 导航；而 app/components/util/Link.vue 给所有
 * isExtLink() 为真的链接都加了 target="_blank"，所以走的就是这条分支。
 * 因此在捕获阶段 preventDefault() 就能取消浏览器原生「开新标签」的默认行为，
 * 不存在与既有 JS 处理器的竞态。
 */
export default defineNuxtPlugin((nuxtApp) => {
	const config = safeGoConfig
	if (!config.enable)
		return

	function findAnchor(event: MouseEvent) {
		for (const node of event.composedPath()) {
			if (node instanceof HTMLAnchorElement)
				return node
		}
		return event.target instanceof Element ? event.target.closest('a[href]') : null
	}

	function isIgnored(anchor: HTMLAnchorElement) {
		return config.links.ignoreAttrs.some(attribute => anchor.hasAttribute(attribute))
	}

	function handle(event: MouseEvent, middleClick: boolean) {
		if (event.defaultPrevented)
			return
		if (middleClick && !config.behavior.interceptModifiedClick)
			return
		const anchor = findAnchor(event)
		if (!anchor || isIgnored(anchor))
			return
		const href = anchor.getAttribute('href')
		if (!href)
			return
		const target = resolveInterceptedHref(href, window.location.href, config.links)
		if (!target)
			return
		const { path } = nuxtApp.$router.currentRoute.value
		// 中转页自身永不拦截，避免配置被改动后出现死循环
		if (normalizeContentPath(path) === normalizeContentPath(config.route.path))
			return
		if (!isPageInScope(path, config.pages))
			return
		// 修饰键点击表达的是「在新标签打开」的意图，拦截时不要吞掉这个意图
		const newTab = middleClick || event.ctrlKey || event.metaKey || event.shiftKey
		event.preventDefault()
		const url = buildSafeGoUrl(config.route.path, config.route.param, target, config.route.base64)
		if (newTab)
			window.open(url, '_blank')
		else
			nuxtApp.$router.push(url)
	}

	const onClick = (event: MouseEvent) => handle(event, false)
	const onAuxClick = (event: MouseEvent) => {
		if (event.button === 1)
			handle(event, true)
	}

	document.addEventListener('click', onClick, true)
	document.addEventListener('auxclick', onAuxClick, true)
	nuxtApp.vueApp.onUnmount(() => {
		document.removeEventListener('click', onClick, true)
		document.removeEventListener('auxclick', onAuxClick, true)
	})
})
```

说明：`resolveInterceptedHref`、`isPageInScope`、`buildSafeGoUrl` 与 `normalizeContentPath` 都来自 `shared/utils/`，已被 Nuxt 自动导入（与现有 `shared/utils/link.ts` 的 `isExtLink` 同机制），无需手写 import。

```ts
})
const countdownText = computed(() => text.countdown.replace('{seconds}', String(Math.max(remaining.value, 0))))

let timer: ReturnType<typeof setInterval> | undefined

function stopTimer() {
	if (timer !== undefined) {
		clearInterval(timer)
		timer = undefined
	}
}

function jump() {
	if (!targetInfo.value)
		return
	stopTimer()
	// 用 replace 而非 href：历史栈里不留中转页，外部站点按返回键直接回到博客原位置
	window.location.replace(targetInfo.value.href)
}

function goBack() {
	stopTimer()
	// 同标签页普通点击路径：Vue Router 会带上 back，Nuxt 据此恢复原来的滚动位置
	if (window.history.state?.back) {
		router.back()
		return
	}
	// 由 window.open 打开的中转页标签可以直接关掉；非脚本打开的标签页浏览器会拒绝 close()
	window.close()
	if (!window.closed)
		navigateTo('/')
}

function startTimer() {
	if (!hasCountdown.value)
		return
	remaining.value = behavior.countdown
	timer = setInterval(() => {
		if (paused.value)
			return
		remaining.value -= 1
		if (remaining.value <= 0)
			jump()
	}, 1000)
}
```

```ts
const router = useRouter()
```

```ts
	if (event.key === 'Escape')
		goBack()
}

onMounted(() => {
	const raw = currentRoute.query[routeConfig.param]
	target.value = resolveTarget(typeof raw === 'string' ? raw : undefined, routeConfig.base64)
	if (targetInfo.value)

	window.addEventListener('keydown', onKeydown)
})

	window.removeEventListener('keydown', onKeydown)
```
